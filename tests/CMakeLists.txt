cmake_minimum_required(VERSION 3.20)
project(sui_tests LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
enable_testing()

add_executable(sui_dpi_tests sui_dpi_tests.c)
target_include_directories(sui_dpi_tests PRIVATE ../include .)
target_compile_definitions(sui_dpi_tests PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(sui_dpi_tests PRIVATE $<$<C_COMPILER_ID:MSVC>:/utf-8>)
target_link_libraries(sui_dpi_tests PRIVATE d2d1 dwrite user32 gdi32 ole32 windowscodecs)
add_test(NAME sui_dpi_tests COMMAND sui_dpi_tests)

add_executable(sui_tests sui_tests.c)
target_include_directories(sui_tests PRIVATE ../include .)
target_compile_definitions(sui_tests PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _CRT_SECURE_NO_WARNINGS)
target_compile_options(sui_tests PRIVATE $<$<C_COMPILER_ID:MSVC>:/utf-8> $<$<C_COMPILER_ID:MSVC>:/W3>)
target_link_libraries(sui_tests PRIVATE d2d1 dwrite user32 gdi32 ole32 windowscodecs)
add_test(NAME sui_tests COMMAND sui_tests)

add_executable(sui_omit_tests sui_omit_tests.c)

target_link_libraries(sui_omit_tests PRIVATE d2d1 dwrite user32 gdi32 ole32 windowscodecs)
add_test(NAME sui_omit_tests COMMAND sui_omit_tests)

add_executable(sui_vsync_tests sui_vsync_tests.c sui_vsync_other.c)
target_include_directories(sui_vsync_tests PRIVATE ../include .)
target_link_libraries(sui_vsync_tests PRIVATE user32 dwmapi)
add_test(NAME sui_vsync_tests COMMAND sui_vsync_tests)
